**核型目标**

对不合格的分镜做修订。



**工作流**

> 见 PPT（P1）



## schema

`revision_record.py`

```py
from pydantic import BaseModel, ConfigDict, Field

class RevisionRecord(BaseModel):
    model_config = ConfigDict(extra="forbid")
    version: int = Field(ge=0)
    fingerprint: str = Field(min_length=16, max_length=16)
    source_codes: list[str]
    target_shot_ids: list[str]
```

- version：版本号
- fingerprint：这一整版分镜的 **<u>指纹</u>**，通过计算 hash 得到。
- source_codes：因为哪类问题而修订，例如 `["plot_gap"]`
- target_shot_ids：这一版实际允许改的镜头 ID，来自报告的 `affected_shot_ids`

## prompt

`revision_prompt.py`

```py
REVISION_PROMPT = """你是漫剧分镜修订者。只根据质量报告修改当前完整分镜，不改需求、故事大纲、角色库或场景库。
输入会明确 allowed_shot_ids；只返回这些目标镜头的完整 StoryBoard 对象，放在 storyboard_list 中，不返回非目标镜头。
必须恰好覆盖全部 allowed_shot_ids，不得遗漏、重复、增加或改写 ID；保持对应 order 不变。
Python 会合并原稿；目标镜头的时长合计必须保持不变，以保持全片总时长。
当 allowed_shot_ids 包含全部镜头时，才允许整稿修订；不能自行扩大范围。
采用解决问题所需的最小改动，优先处理 error；warning 仅在不引入新矛盾且不扩大范围时顺带改善。保留目标镜头中与问题无关的原有内容，不做全段风格重写。
逐条核对 description 与 suggestion，在原有设定下落实修复；建议是修复线索，不可机械照抄与原稿冲突的建议。
不引入无关道具、额外角色动作、新奇幻规则或新的情节分支来解释旧问题；补充必要过渡应使用已有角色、道具与设定。
核对道具来源、数量、持有者与状态，以及入口、人物的空间关系；目标镜头必须与未修改的前后镜头衔接。
同步调整目标镜头中确有必要的 action、dialogue、首尾帧和 video_prompt，避免只补一个字段却与其他字段冲突；人物运动与摄影机运动分开描述。
若问题依赖不可修改的上游设定或非目标镜头，不擅自扩大修订范围或捏造规则；保留无法安全修改的内容，留给后续评价与人工处理。
只引用已有角色和场景 ID。不要添加解释文本，只返回符合 StoryBoardList Schema 的结构化对象，镜头放在 storyboard_list 中。"""

```

首先在实例化 `FakeTextModel` 模型的时候，又多了一个实例参数：`_revision_scenario`

该参数的值来自于环境变量 `FAKE_REVISION_SCENARIO`，这个值用于模拟分镜修订时的不同情况：

- `none`：按质量报告改目标镜头，产生新版本。
- `second_pass`：首评失败，修订后复评通过。
- `always_fail`：每次评价都不通过，改满次数后 revision_exhausted。
- `unchanged`：修订内容不变，停在 revision_stalled / no_change。
- `hard_regression`：修订后引用不存在的角色，再撞硬规则。
- `repeat_version`：第二次修订退回初版，停在 revision_stalled / repeated_version。

有一个注意点：

1. second_pass / always_fail 主要改变复评结果
2. 其余改变改稿内容

**新增修订分镜逻辑**

```py
def revise_storyboard(
        self,
        requirement: CreativeRequirement,
        outline: StoryOutline,
        characters: list[CharacterProfile],
        scenes: list[SceneProfile],
        storyboard: list[StoryBoard],
        report: StoryboardQualityReport,
        revision_count: int = 0,
    ) -> list[StoryBoard]:
        if self._revision_scenario == "unchanged":
            return [shot.model_copy(deep=True) for shot in storyboard]

        if self._revision_scenario == "repeat_version" and revision_count >= 1:
            return [_undo_fake_revision(shot) for shot in storyboard]

        revised = [shot.model_copy(deep=True) for shot in storyboard]

        if len({shot.shot_id for shot in revised}) != len(revised):
            revised = [
                shot.model_copy(update={"shot_id": f"shot_{shot.order:02d}"})
                for shot in revised
            ]

        if (
            sum(shot.duration_seconds for shot in revised)
            != requirement.duration_seconds
        ):
            remainder = requirement.duration_seconds - sum(
                shot.duration_seconds for shot in revised[:-1]
            )
            revised[-1] = revised[-1].model_copy(update={"duration_seconds": remainder})

        target_ids = report.affected_shot_ids or [revised[0].shot_id]
        target_id = target_ids[0]
        index_by_id = {shot.shot_id: index for index, shot in enumerate(revised)}

        if target_id not in index_by_id:
            raise ValueError(f"Fake 修订目标不存在：{target_id}")

        target_index = index_by_id[target_id]
        target = revised[target_index]

        if self._revision_scenario == "hard_regression":
            revised[target_index] = target.model_copy(
                update={"character_ids": ["char_missing"]}
            )
        else:
            revised[target_index] = target.model_copy(
                update={
                    "action": f"{_ACTION_PREFIX}{target.action}",
                    "character_ids": ["char_xiaolin", "char_amu"],
                    "start_frame_prompt": f"{target.start_frame_prompt}{_FRAME_SUFFIX}",
                    "end_frame_prompt": f"{target.end_frame_prompt}{_FRAME_SUFFIX}",
                }
            )
        
        return revised
```

在做 **修订把稿改回了初版，指纹撞上历史记录** 的模拟的时候，涉及到一个辅助方法：

```py
_ACTION_PREFIX = "已按质量报告修订："
_FRAME_SUFFIX = "，已按反馈强化"

def _undo_fake_revision(shot: StoryBoard) -> StoryBoard:
    action = shot.action
    action = action.removeprefix(_ACTION_PREFIX)
    start = shot.start_frame_prompt.removesuffix(_FRAME_SUFFIX)
    end = shot.end_frame_prompt.removesuffix(_FRAME_SUFFIX)
    return shot.model_copy(
        update={
            "action": action,
            "start_frame_prompt": start,
            "end_frame_prompt": end,
        }
    )
```

### 3. 真实模型

`ark_text_model.py`

```py
def revise_storyboard(
    self,
    requirement: CreativeRequirement,
    outline: StoryOutline,
    characters: list[CharacterProfile],
    scenes: list[SceneProfile],
    storyboard: list[StoryBoard],
    report: StoryboardQualityReport,
    revision_count: int = 0,
) -> list[StoryBoard]:
    old_by_id = {shot.shot_id: shot for shot in storyboard}
    
    if len(old_by_id) != len(storyboard):
        raise ValueError("真实局部修订要求原稿 shot_id 唯一，先处理硬规则错误")
        
    targets = set(report.affected_shot_ids) or set(old_by_id)
    unknown = targets - set(old_by_id)
    
    if unknown:
        raise ValueError(f"修订目标引用未知分镜：{sorted(unknown)}")
        
    allowed_ids = [shot.shot_id for shot in storyboard if shot.shot_id in targets]
    
    result = self._parse(
        system=REVISION_PROMPT
        + "\n本次唯一允许返回的镜头 ID："
        + json.dumps(allowed_ids, ensure_ascii=False),
        payload={
            "requirement": requirement.model_dump(mode="json"),
            "story_outline": outline.model_dump(mode="json"),
            "characters": [item.model_dump(mode="json") for item in characters],
            "scenes": [item.model_dump(mode="json") for item in scenes],
            "storyboard": [item.model_dump(mode="json") for item in storyboard],
            "quality_report": report.model_dump(mode="json"),
            "allowed_shot_ids": allowed_ids,
        },
        schema=StoryBoardList,
    )
    
    revised = [StoryBoard.model_validate(shot) for shot in result.storyboard_list]
    returned_ids = [shot.shot_id for shot in revised]
    
    if len(set(returned_ids)) != len(returned_ids) or set(returned_ids) != targets:
        raise ValueError("修订输出必须恰好覆盖允许的镜头 ID，不能遗漏、重复或越界")
        
    revised_by_id = {shot.shot_id: shot for shot in revised}
    
    if any(shot.order != old_by_id[shot.shot_id].order for shot in revised):
        raise ValueError("修订不得改变目标镜头 order")
        
    if sum(shot.duration_seconds for shot in revised) != sum(
        old_by_id[shot_id].duration_seconds for shot_id in targets
    ):
        raise ValueError("修订不得改变目标镜头时长合计")

    return [revised_by_id.get(shot.shot_id, shot) for shot in storyboard]
```

该节点还涉及到一个来自于 `shot_fingerprint.py` 的辅助方法：

```py
import hashlib
import json

from server.schema import StoryBoard


def shot_fingerprint(shots: list[StoryBoard]) -> str:
    """对完整分镜取稳定指纹，用于识别未变化或回到旧版本。"""
    payload = [shot.model_dump(mode="json") for shot in shots]
    raw = json.dumps(payload, ensure_ascii=False, sort_keys=True).encode("utf-8")
    return hashlib.sha256(raw).hexdigest()[:16]
  
```

### make_revise_storyboard

`make_revise_storyboard.py`

```py
from server.model import TextModel
from server.schema import RevisionRecord, StoryBoard
from server.states import ComicDramaState
from .shot_fingerprint import shot_fingerprint

def make_revise_storyboard(text_model: TextModel):
    def revise_storyboard(state: ComicDramaState) -> dict[str, object]:
        report = state["quality_report"]
        
        if report is None or report.is_passed or not report.issues:
            raise ValueError("没有可执行的质量问题，不能修订")
            
        if state["revision_count"] >= state["max_revisions"]:
            raise ValueError("修订次数已达上限")

        old_shots = state["storyboard"]
        requirement = state["requirement"]
        candidate = [
            StoryBoard.model_validate(shot)
            for shot in text_model.revise_storyboard(
                requirement, state["story_outline"],
                state["characters"], state["scenes"], old_shots, report,
                revision_count=state["revision_count"],
            )
        ]

        old_by_order = {shot.order: shot for shot in old_shots}
        new_by_order = {shot.order: shot for shot in candidate}
        old_ids = [shot.shot_id for shot in old_shots]
        targets = set(report.affected_shot_ids)
        
        if len(candidate) != requirement.shot_count:
            raise ValueError("修订不得改变镜头数量")
            
        if sum(shot.duration_seconds for shot in candidate) != requirement.duration_seconds:
            raise ValueError("修订不得改变目标总时长")
            
        if new_by_order.keys() != old_by_order.keys():
            raise ValueError("修订不得改变镜头顺序号")
            
        if len(set(old_ids)) == len(old_ids) and [s.shot_id for s in candidate] != old_ids:
            raise ValueError("已有唯一 shot_id 必须保持稳定")
            
        if targets and any(
            old_by_order[order] != shot and old_by_order[order].shot_id not in targets
            for order, shot in new_by_order.items()
        ):
            raise ValueError("修订改动了报告未指出的镜头")

        signature = shot_fingerprint(candidate)
        history = state["revision_history"]
        if signature in {record.fingerprint for record in history}:
            reason = "no_change" if signature == shot_fingerprint(old_shots) else "repeated_version"
            return {
                "stop_reason": reason,
                "current_stage": "revision_stalled",
                "events": ["修订没有产生新版本，工作流停止"],
            }

        next_version = state["revision_count"] + 1
        return {
            "storyboard": candidate,
            "quality_report": None,
            "previous_quality_report": report.model_dump(mode="json"),
            "storyboard_validation": None,
            "revision_count": next_version,
            "revision_history": [*history, RevisionRecord(
                version=next_version, fingerprint=signature,
                source_codes=sorted({issue.code for issue in report.issues}),
                target_shot_ids=sorted(targets),
            )],
            "stop_reason": None,
            "current_stage": "storyboard_revised",
            "events": [f"已生成第 {next_version} 版修订分镜，等待重新校验"],
        }

    return revise_storyboard
  
```
